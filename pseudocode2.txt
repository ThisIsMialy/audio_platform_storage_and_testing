    Function HashFunc(STRING key)
        integer index

        For (i ← 0 Until i is less than key.size())
            sum ← sum + (ASCII of key[i] * prime) Mod this.capacity)
        Endfor
        index ← sum;
        Return index;
    endfunction

        Return this.buckets / this.capacity;
